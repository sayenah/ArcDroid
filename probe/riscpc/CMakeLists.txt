cmake_minimum_required(VERSION 3.22)
project(arcdroid_rpcprobe C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(PROBE_SANITIZERS "ASan and UBSan on host" OFF)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
set(UP ${CMAKE_CURRENT_SOURCE_DIR}/rpcemu-0.9.5/src)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
 "${CMAKE_CURRENT_SOURCE_DIR}/boot/Multisync" "${CMAKE_CURRENT_SOURCE_DIR}/boot/!Boot" "${CMAKE_CURRENT_SOURCE_DIR}/boot/BootEnv" "${CMAKE_CURRENT_SOURCE_DIR}/boot/BootTasks")
foreach(asset MDF BOOT_OBEY BOOT_ENV BOOT_TASKS)
 if(asset STREQUAL "MDF")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/boot/Multisync" PROBE_${asset})
 elseif(asset STREQUAL "BOOT_OBEY")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/boot/!Boot" PROBE_${asset})
 elseif(asset STREQUAL "BOOT_ENV")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/boot/BootEnv" PROBE_${asset})
 else()
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/boot/BootTasks" PROBE_${asset})
 endif()
 string(REPLACE "\\" "\\\\" PROBE_${asset} "${PROBE_${asset}}")
 string(REPLACE "\"" "\\\"" PROBE_${asset} "${PROBE_${asset}}")
 if(asset STREQUAL "MDF")
  string(REPLACE "\n" "\\n\"\n\"" PROBE_${asset} "${PROBE_${asset}}")
 else()
  string(REPLACE "\n" "\\r\"\n\"" PROBE_${asset} "${PROBE_${asset}}")
 endif()
endforeach()
configure_file(src/boot_assets.h.in boot_assets.h @ONLY)
add_compile_options(-ffunction-sections -fdata-sections)
if(APPLE)
 add_link_options(-Wl,-dead_strip)
else()
 add_link_options(-Wl,--gc-sections)
endif()
add_library(probe_zlib STATIC
 ${ROOT}/third_party/zlib/adler32.c ${ROOT}/third_party/zlib/crc32.c
 ${ROOT}/third_party/zlib/inffast.c ${ROOT}/third_party/zlib/inflate.c
 ${ROOT}/third_party/zlib/inftrees.c ${ROOT}/third_party/zlib/zutil.c)
target_include_directories(probe_zlib PUBLIC ${ROOT}/third_party/zlib)
target_compile_definitions(probe_zlib PRIVATE Z_SOLO)
target_compile_options(probe_zlib PRIVATE -fvisibility=hidden)
add_library(probe_engine STATIC
  ${UP}/arm.c ${UP}/arm_common.c ${UP}/cp15.c ${UP}/mem.c
  ${UP}/iomd.c ${UP}/vidc20.c ${UP}/keyboard.c ${UP}/cmos.c
  ${UP}/superio.c ${UP}/i8042.c ${UP}/fdc.c src/media.c src/jfd_presentation.c src/filecore_limits.c
  src/machine.c src/mode_observer.c src/boot_helper.c src/sound.c src/absent_devices.c src/ide_probe.c ${ROOT}/core/hash.c
  ${ROOT}/core/jfd.c ${ROOT}/core/jfd_layout.c ${ROOT}/core/jfd_file_data.c
  ${ROOT}/core/launch_discovery.c src/install_disc.c)
target_include_directories(probe_engine PUBLIC src ${UP} ${ROOT}/core ${ROOT}/libretro/include ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(probe_engine PUBLIC probe_zlib)
target_compile_definitions(probe_engine PUBLIC ARCDROID_PROBE=1 RELEASE_BUILD=1)
target_compile_options(probe_engine PUBLIC -Wall -Wextra -fno-strict-aliasing -fvisibility=hidden)
if(PROBE_SANITIZERS)
 target_compile_options(probe_engine PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(probe_engine PUBLIC -fsanitize=address,undefined)
endif()
add_library(arcdroid_rpcprobe_libretro SHARED src/libretro.c src/frontend.c)
set_target_properties(arcdroid_rpcprobe_libretro PROPERTIES PREFIX "")
target_link_libraries(arcdroid_rpcprobe_libretro PRIVATE probe_engine)
if(ANDROID)
 target_link_options(arcdroid_rpcprobe_libretro PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--version-script=${ROOT}/libretro/link.T")
 target_link_libraries(arcdroid_rpcprobe_libretro PRIVATE m dl)
elseif(WIN32)
 target_link_options(arcdroid_rpcprobe_libretro PRIVATE -static-libgcc)
endif()
add_executable(probe_mechanisms src/mechanisms.c)
target_link_libraries(probe_mechanisms PRIVATE probe_engine)
add_executable(probe_runner src/runner.c)
target_include_directories(probe_runner PRIVATE ${ROOT}/libretro/include)
if(NOT WIN32)
 target_link_libraries(probe_runner PRIVATE ${CMAKE_DL_LIBS})
endif()
enable_testing()
add_test(NAME mechanisms COMMAND probe_mechanisms)

if(PROBE_SANITIZERS)
 target_compile_options(probe_runner PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(probe_runner PRIVATE -fsanitize=address,undefined)
endif()
add_executable(probe_lifecycle src/lifecycle.c)
target_link_libraries(probe_lifecycle PRIVATE arcdroid_rpcprobe_libretro)
target_include_directories(probe_lifecycle PRIVATE ${ROOT}/libretro/include)
if(PROBE_SANITIZERS)
 target_compile_options(probe_lifecycle PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(probe_lifecycle PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME lifecycle COMMAND probe_lifecycle)
add_executable(probe_controller src/controller.c)
target_link_libraries(probe_controller PRIVATE probe_engine)
add_test(NAME controller COMMAND probe_controller)
add_executable(probe_frontend src/frontend_test.c src/frontend.c)
target_link_libraries(probe_frontend PRIVATE probe_engine)
add_test(NAME frontend COMMAND probe_frontend)
add_executable(probe_ide_controller src/ide_controller.c)
target_link_libraries(probe_ide_controller PRIVATE probe_engine)
